# Windows wxWidgets UI builds by default. CI can explicitly disable it for headless checks.
if(BRASSCUE_BUILD_GUI)
  set(brasscue_wx_components core base)
  set(brasscue_web_source gui/web_panel_stub.cpp)
  if(BRASSCUE_ENABLE_WEB_PANEL)
    list(APPEND brasscue_wx_components webview)
    set(brasscue_web_source gui/web_panel.cpp)
  endif()
  if(WIN32)
    find_package(wxWidgets CONFIG QUIET COMPONENTS ${brasscue_wx_components})
  else()
    find_package(wxWidgets QUIET COMPONENTS ${brasscue_wx_components})
  endif()
  if(NOT wxWidgets_FOUND OR (WIN32 AND BRASSCUE_ENABLE_WEB_PANEL AND NOT TARGET wx::webview))
    message(FATAL_ERROR "Required wxWidgets components (${brasscue_wx_components}) are missing. Classic Windows vcpkg: vcpkg install wxwidgets[webview]:x64-windows --classic --recurse; configure with VCPKG_MANIFEST_MODE=OFF and x64-windows. Install Microsoft Edge WebView2 Evergreen Runtime. Linux: install libwxgtk3.2-dev and libwxgtk-webview3.2-dev. Disable BRASSCUE_ENABLE_WEB_PANEL for the native GUI.")
  endif()
  if(NOT WIN32)
    include(${wxWidgets_USE_FILE})
  endif()
  add_library(brasscue-gui-support
    gui/widgets.cpp gui/profile_editor.cpp gui/profile_files.cpp
    gui/assignment_table.cpp gui/button_editor.cpp gui/dark_theme.cpp
    gui/network_panel.cpp gui/native_panel.cpp gui/panel_host.cpp ${brasscue_web_source})
  target_compile_features(brasscue-gui-support PUBLIC cxx_std_17)
  target_compile_definitions(brasscue-gui-support PUBLIC NOMINMAX BRASSCUE_ENABLE_WEB_PANEL=$<BOOL:${BRASSCUE_ENABLE_WEB_PANEL}>)
  target_link_libraries(brasscue-gui-support PUBLIC brasscue-profile brasscue-network brasscue-presentation)
  if(WIN32)
    target_link_libraries(brasscue-gui-support PUBLIC wx::core wx::base ws2_32 dwmapi)
    if(BRASSCUE_ENABLE_WEB_PANEL)
      target_link_libraries(brasscue-gui-support PUBLIC wx::webview)
    endif()
  else()
    target_link_libraries(brasscue-gui-support PUBLIC ${wxWidgets_LIBRARIES})
  endif()
  add_executable(brasscue-gui WIN32 gui/main.cpp gui/main_frame.cpp)
  target_link_libraries(brasscue-gui PRIVATE brasscue-gui-support)
  if(BRASSCUE_ENABLE_WEB_PANEL)
    function(brasscue_copy_web_assets target)
      add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${target}>/web_assets"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${CMAKE_CURRENT_SOURCE_DIR}/gui/web_assets/compact.html"
          "${CMAKE_CURRENT_SOURCE_DIR}/gui/web_assets/compact.css"
          "${CMAKE_CURRENT_SOURCE_DIR}/gui/web_assets/compact.js"
          "$<TARGET_FILE_DIR:${target}>/web_assets")
    endfunction()
    brasscue_copy_web_assets(brasscue-gui)
    install(FILES gui/web_assets/compact.html gui/web_assets/compact.css gui/web_assets/compact.js DESTINATION bin/web_assets)
  endif()
  install(TARGETS brasscue-gui RUNTIME DESTINATION bin)
endif()
