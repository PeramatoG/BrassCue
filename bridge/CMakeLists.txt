# Windows wxWidgets UI builds by default. CI can explicitly disable it for headless checks.
if(BRASSCUE_BUILD_GUI)
  if(WIN32)
    find_package(wxWidgets CONFIG REQUIRED COMPONENTS core base)
  else()
    find_package(wxWidgets REQUIRED COMPONENTS core base)
    include(${wxWidgets_USE_FILE})
  endif()
  add_executable(brasscue-gui WIN32 gui/main.cpp gui/main_frame.cpp gui/widgets.cpp gui/profile_editor.cpp gui/profile_files.cpp)
  target_compile_features(brasscue-gui PRIVATE cxx_std_17)
  target_compile_definitions(brasscue-gui PRIVATE NOMINMAX)
  target_link_libraries(brasscue-gui PRIVATE brasscue-profile)
  if(WIN32)
    target_link_libraries(brasscue-gui PRIVATE wx::core wx::base ws2_32)
  else()
    target_link_libraries(brasscue-gui PRIVATE ${wxWidgets_LIBRARIES})
  endif()
endif()
